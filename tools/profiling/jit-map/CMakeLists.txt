cmake_minimum_required(VERSION 3.16)
project(EcoJitMap CXX)
if(NOT AR_WINDOWS OR NOT EXISTS "${ECO_JDK}/include/jvmti.h")
    message(FATAL_ERROR "Use the Windows toolchain and -DECO_JDK=<JDK directory>")
endif()
add_library(EcoJitMap SHARED agent.cpp)
target_compile_features(EcoJitMap PRIVATE cxx_std_20)
target_include_directories(EcoJitMap PRIVATE "${ECO_JDK}/include" "${ECO_JDK}/include/win32")
target_compile_definitions(EcoJitMap PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(EcoJitMap PRIVATE /MT /EHsc /utf-8)
